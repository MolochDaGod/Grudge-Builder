        function updateLevel() {
            // Calculate XP loss since last interaction
            const now = Date.now();
            const elapsed = (now - gameState.stats.lastInteractionTime) / 1000; // Time in seconds
            const rate = 0.05735 + 0.000572 * (gameState.stats.level - 1); // XP loss rate per second

            gameState.stats.xp = Math.max(0, gameState.stats.xp - xpLoss);

            // Determine level based on XP
            let level = 1;
            while (level < 99 && gameState.stats.xp >= xpThresholds[level + 1]) {
                level++;
            }
            gameState.stats.level = level;

            // Update last interaction time only if an action occurs; otherwise, keep it for next calculation

        }

                }
            });
            inv.crafted.forEach((item, i) => {
                const div = document.createElement('div');
                div.className = 'resource-item bg-gray-700 p-2 rounded-lg text-center';
                div.innerHTML = `<p class="text-sm">${item}</p>`;
                invDiv.appendChild(div);
            });

            // Gear
            ['weapon', 'utility', 'armor', 'tool1', 'heal'].forEach(slot => {
                const gearDiv = document.getElementById(`equipped-${slot}`);
                gearDiv.innerHTML = inv.gear[slot] ? inv.gear[slot] : 'None';
            });

            // Crafting List (Original Items)
            const craftListDiv = document.getElementById('crafting-list');
            craftListDiv.innerHTML = '';
            originalCraftables.forEach(item => {
                const canCraft = Object.keys(item.req).every(r => gameState.inventory[r] >= item.req[r]);
                const div = document.createElement('div');
                div.className = `bg-gray-700 p-3 rounded-lg ${canCraft ? '' : 'opacity-50'}`;
                div.innerHTML = `
                    <p class="text-blue-300 text-sm">${item.name} (${Object.entries(item.req).map(([k, v]) => `${k} × ${v}`).join(', ')}) - ${item.time}s</p>
                    <button class="bg-blue-600 text-white px-3 py-1 rounded mt-2 w-full hover:bg-blue-700 transition text-sm ${canCraft ? '' : 'cursor-not-allowed bg-gray-600'}" 
                            onclick="craftItem('Crafting', '${item.name}', ${JSON.stringify(item.req)}, ${item.time}, '${item.type}')" ${canCraft ? '' : 'disabled'}>Craft</button>
                `;
                craftListDiv.appendChild(div);
            });

            // Crafting Stations
            const benchesDiv = document.getElementById('benches');
            benchesDiv.innerHTML = '';
            Object.keys(recipes).forEach(structure => {
                if (gameState.cityGrid.includes(structure)) {
                    const div = document.createElement('div');
                    div.className = 'bg-gray-700 p-2 rounded-lg';
                    div.innerHTML = `<p class="text-blue-300 text-sm">${structure}</p><button class="bg-blue-600 text-white px-2 py-1 rounded mt-1 w-full hover:bg-blue-700 transition text-sm" onclick="showCrafting('${structure}')">Use</button>`;
                    benchesDiv.appendChild(div);
                }
            });

            // City Grid
            const gridDiv = document.getElementById('city-grid');
            gridDiv.innerHTML = '';
            gameState.cityGrid.forEach((structure, i) => {
                const div = document.createElement('div');
                div.className = `grid-square ${structure ? 'occupied' : ''}`;
                div.textContent = structure || 'Empty';
                if (structure) div.onclick = () => showCrafting(structure);
                gridDiv.appendChild(div);
            });

            // City Inventory
            const cityInvDiv = document.getElementById('city-inventory');
            cityInvDiv.innerHTML = '';
            inv.structures.forEach((item, i) => {
                const div = document.createElement('div');
                div.className = 'bg-gray-700 p-2 rounded-lg';
                div.innerHTML = `<p class="text-sm">${item}</p><button class="bg-teal-600 text-white px-2 py-1 rounded mt-1 w-full hover:bg-teal-700 transition text-sm" onclick="placeStructure(${i})">Place</button>`;
                cityInvDiv.appendChild(div);
            });

            // Sell List
            const sellListDiv = document.getElementById('sell-list');
            sellListDiv.innerHTML = '';
            Object.keys(sellPrices).forEach(resource => {
                if (inv[resource] > 0) {
                    const div = document.createElement('div');
                    div.className = 'bg-gray-700 p-2 rounded-lg';
                    div.innerHTML = `<p class="text-yellow-300 text-sm">${resource}: ${inv[resource]}</p><button class="bg-yellow-600 text-white px-2 py-1 rounded mt-1 w-full hover:bg-yellow-700 transition text-sm" onclick="sellResource('${resource}')">Sell 1</button>`;
                    sellListDiv.appendChild(div);
                }
            });

            // Buy List
            const buyListDiv = document.getElementById('buy-list');
            buyListDiv.innerHTML = '';
            buyItems.forEach(item => {
                const div = document.createElement('div');
                div.className = 'bg-gray-700 p-2 rounded-lg';
                div.innerHTML = `<p class="text-yellow-300 text-sm">${item.name} - ${item.cost}g</p><button class="bg-yellow-600 text-white px-2 py-1 rounded mt-1 w-full hover:bg-yellow-700 transition text-sm" onclick="buyItem('${item.name}')">Buy</button>`;
                buyListDiv.appendChild(div);
            });

            // Battle List
            const battleListDiv = document.getElementById('battle-list');
            battleListDiv.innerHTML = '';
            inv.battleItems.forEach((item, i) => {
                const div = document.createElement('div');
                div.className = 'bg-gray-700 p-2 rounded-lg';
                div.innerHTML = `<p class="text-red-300 text-sm">${item.name}</p><button class="bg-red-600 text-white px-2 py-1 rounded mt-1 w-full hover:bg-red-700 transition text-sm" onclick="engageBattle(${i})">Fight</button>`;
                battleListDiv.appendChild(div);
            });
        }

        // Gathering
        function gatherResource(resource) {
            const config = gatheringConfig[resource];
            const energyCost = config.tool && gameState.inventory.gear.tool1 === config.tool ? config.toolEnergy : config.baseEnergy;
            if (!gameState.cooldowns[resource] && gameState.stats.energy >= energyCost) {
                gameState.stats.energy -= energyCost;
                gameState.inventory[resource] += 1;
                gameState.stats.lastInteractionTime = Date.now(); // Update interaction time
                startCooldown(resource, config.baseCooldown);
                updateUI();
            } else if (gameState.stats.energy < energyCost) {
                alert('Not enough energy!');
            }
        }

        function startCooldown(resource, duration) {
            const button = document.getElementById(`gather-${resource}`);
            const cooldownText = document.getElementById(`${resource}-cooldown`);
            gameState.cooldowns[resource] = true;
            button.disabled = true;
            button.classList.remove('pulse', 'bg-purple-600', 'hover:bg-purple-700');
            button.classList.add('bg-gray-600');
            let timeLeft = duration;
            cooldownText.textContent = `${timeLeft}s`;
            const interval = setInterval(() => {
                timeLeft--;
                cooldownText.textContent = timeLeft > 0 ? `${timeLeft}s` : 'Ready';
                if (timeLeft <= 0) {
                    clearInterval(interval);
                    gameState.cooldowns[resource] = false;
                    button.disabled = false;
                    button.classList.add('pulse', 'bg-purple-600', 'hover:bg-purple-700');
                    button.classList.remove('bg-gray-600');
                }
            }, 1000);
        }

        // Crafting
        function showCrafting(structure) {
            const modal = document.getElementById('modal');
            const title = document.getElementById('modal-title');
            const body = document.getElementById('modal-body');
            title.textContent = `${structure} Crafting`;
            body.innerHTML = '';
            recipes[structure].forEach(item => {
                const canCraft = Object.keys(item.req).every(r => gameState.inventory[r] >= item.req[r]);
                const div = document.createElement('div');
                div.className = `bg-gray-700 p-3 rounded-lg ${canCraft ? '' : 'opacity-50'}`;
                div.innerHTML = `
                    <p class="text-blue-300 text-sm">${item.name} (${Object.entries(item.req).map(([k, v]) => `${k} × ${v}`).join(', ')}) - ${item.time}s</p>
                    <button class="bg-blue-600 text-white px-3 py-1 rounded mt-2 w-full hover:bg-blue-700 transition text-sm ${canCraft ? '' : 'cursor-not-allowed bg-gray-600'}" 
                            onclick="craftItem('${structure}', '${item.name}', ${JSON.stringify(item.req)}, ${item.time}, '${item.type}')" ${canCraft ? '' : 'disabled'}>Craft</button>
                `;
                body.appendChild(div);
            });
            modal.style.display = 'block';
        }

        function craftItem(structure, name, requirements, time, type) {
            if (gameState.stats.energy >= 5 && Object.keys(requirements).every(r => gameState.inventory[r] >= requirements[r])) {
                gameState.stats.energy -= 5;
                Object.keys(requirements).forEach(r => gameState.inventory[r] -= requirements[r]);
                setTimeout(() => {
                    if (type === 'resource') {
                        gameState.inventory[name.toLowerCase()] = (gameState.inventory[name.toLowerCase()] || 0) + 1;
                    } else if (type === 'structure') {
                        gameState.inventory.structures.push(name);
                    } else {
                        gameState.inventory.crafted.push(name);
                    }
                    gameState.stats.xp += 10;
                    gameState.stats.lastInteractionTime = Date.now(); // Update interaction time
                    updateUI();
                    alert(`Crafted ${name}!`);
                }, time * 1000);
                alert(`Crafting ${name}... (${time}s)`);
                if (structure !== 'Crafting') document.getElementById('modal').style.display = 'none';
            } else {
                alert('Not enough resources or energy!');
            }
        }

        // City
        function placeStructure(index) {
            const structure = gameState.inventory.structures[index];
            const emptyIndex = gameState.cityGrid.indexOf(null);
            if (emptyIndex !== -1) {
                gameState.cityGrid[emptyIndex] = structure;
                gameState.inventory.structures.splice(index, 1);
                gameState.stats.lastInteractionTime = Date.now(); // Update interaction time
                updateUI();
                alert(`${structure} placed on the grid!`);
            } else {
                alert('No empty spaces in the city grid!');
            }
        }

        // Eating
        function eatItem(item) {
            if (confirm(`Do you want to eat this ${item}?`)) {
                if (gameState.inventory[item] > 0) {
                    gameState.inventory[item]--;
                    if (item === 'fish') gameState.stats.hunger = Math.min(100, gameState.stats.hunger + 15);
                    if (item === 'berry') gameState.stats.hunger = Math.min(100, gameState.stats.hunger + 10);
                    gameState.stats.lastInteractionTime = Date.now(); // Update interaction time
                    updateUI();
                    alert(`Ate ${item}!`);
                }
            }
        }

        // Equipping
        function equipItem(slot) {
            if (gameState.inventory.crafted.length > 0) {
                const item = gameState.inventory.crafted.pop();
                const validItems = {
                    weapon: ['Spear', 'Sword', 'Steel Sword', 'Dagger', 'Bone Knife', 'Chain Whip'],
                    utility: ['Shield', 'Fish Trap', 'Hemp Net'],
                    armor: ['Metal Armor', 'Steel Armor', 'Hide Armor'],
                    tool1: ['Fishing Pole', 'Hatchet', 'Mining Pick', 'Tanning Knife'],
                    heal: ['Bandage', 'Healing Salve', 'Healing Bandage']
                };
                if (validItems[slot].includes(item)) {
                    gameState.inventory.gear[slot] = item;
                    gameState.stats.lastInteractionTime = Date.now(); // Update interaction time
                    updateUI();
                    alert(`Equipped ${item} in ${slot}!`);
                } else {
                    gameState.inventory.crafted.push(item);
                    alert(`Cannot equip ${item} in ${slot}!`);
                }
            } else {
                alert('No items to equip!');
            }
        }

        // Using Items
        function useItem() {
            const inv = gameState.inventory;
            if (inv.gear.heal) {
                let healAmount = 0;
                if (inv.gear.heal === 'Bandage') healAmount = 10;
                else if (inv.gear.heal === 'Healing Salve') healAmount = 20;
                else if (inv.gear.heal === 'Healing Bandage') healAmount = 30;
                gameState.stats.hp = Math.min(100, gameState.stats.hp + healAmount);
                const usedItem = inv.gear.heal;
                inv.gear.heal = null;
                gameState.stats.lastInteractionTime = Date.now(); // Update interaction time
                updateUI();
                alert(`Used ${usedItem}! Restored ${healAmount} HP.`);
            } else if (inv.gear.weapon) {
                alert(`Swung ${inv.gear.weapon} at an imaginary foe!`);
            } else {
                alert('No usable items equipped!');
            }
        }

        // Selling/Buying
        function sellResource(resource) {
            if (gameState.inventory[resource] > 0) {
                gameState.inventory[resource]--;
                gameState.stats.gold += sellPrices[resource];
                gameState.stats.lastInteractionTime = Date.now(); // Update interaction time
                updateUI();
                alert(`Sold 1 ${resource} for ${sellPrices[resource]} gold!`);
            }
        }

        function buyItem(itemName) {
            const item = buyItems.find(i => i.name === itemName);
            if (gameState.stats.gold >= item.cost) {
                gameState.stats.gold -= item.cost;
                if (item.type === 'food') {
                    gameState.stats.hunger = Math.min(100, gameState.stats.hunger + item.effect.hunger);
                } else if (item.type === 'heal') {
                    gameState.stats.hp = Math.min(100, gameState.stats.hp + item.effect.hp);
                } else if (item.type === 'battle') {
                    gameState.inventory.battleItems.push(item);
                }
                gameState.stats.lastInteractionTime = Date.now(); // Update interaction time
                updateUI();
                alert(`Bought ${item.name}!`);
            } else {
                alert('Not enough gold!');
            }
        }

        // Battles
        function engageBattle(index) {
            const item = gameState.inventory.battleItems[index];
            const enemy = item.enemy;
            const playerStats = getPlayerStats();
            const winChance = 50 + (playerStats.attack - enemy.defend) * 2 + (playerStats.defend - enemy.attack) * 2;
            const winChanceClamped = Math.max(0, Math.min(100, winChance));

            const modal = document.getElementById('modal');
            const title = document.getElementById('modal-title');
            const body = document.getElementById('modal-body');
            title.textContent = `${enemy.name} Encounter`;
            body.innerHTML = `
                <p class="text-gray-300">A ${enemy.name.toLowerCase()} emerges! "${enemy.name === 'Goblin' ? 'It eyes your loot greedily.' : 'It roars a challenge!'} "</p>
                <p class="text-gray-300">Your Stats: Attack ${playerStats.attack}, Defend ${playerStats.defend}</p>
                <p class="text-gray-300">Enemy Stats: Attack ${enemy.attack}, Defend ${enemy.defend}</p>
                <p class="text-gray-300">Win Chance: ${winChanceClamped.toFixed(2)}%</p>
                <button class="bg-red-600 text-white px-3 py-1 rounded mt-2 hover:bg-red-700 transition" onclick="fightEnemy(${index}, ${winChanceClamped})">Battle</button>
            `;
            modal.style.display = 'block';
        }

        function fightEnemy(index, winChance) {
            const random = Math.random() * 100;
            const enemy = gameState.inventory.battleItems[index].enemy;
            if (random < winChance) {
                gameState.stats.xp += enemy.level * 20;
                gameState.stats.gold += enemy.level * 10;
                alert(`Victory! Defeated the ${enemy.name}! Gained ${enemy.level * 20} XP and ${enemy.level * 10} gold.`);
            } else {
                gameState.stats.hp = Math.max(0, gameState.stats.hp - 10);
                alert(`Defeat! The ${enemy.name} overwhelmed you. Lost 10 HP.`);
            }
            gameState.inventory.battleItems.splice(index, 1);
            gameState.stats.lastInteractionTime = Date.now(); // Update interaction time
            updateUI();
            document.getElementById('modal').style.display = 'none';
        }

        // Tab Switching
        const tabs = {
            gather: { tab: document.getElementById('gather-tab'), btn: document.getElementById('tab-gather') },
            craft: { tab: document.getElementById('craft-tab'), btn: document.getElementById('tab-craft') },
            city: { tab: document.getElementById('city-tab'), btn: document.getElementById('tab-city') },
            character: { tab: document.getElementById('character-tab'), btn: document.getElementById('tab-character') },
            sell: { tab: document.getElementById('sell-tab'), btn: document.getElementById('tab-sell') }
        };
        Object.keys(tabs).forEach(key => {
            tabs[key].btn.addEventListener('click', () => {
                Object.values(tabs).forEach(t => {
                    t.tab.classList.add('hidden');
                    t.btn.classList.remove('active', 'bg-amber-600');
                    t.btn.classList.add('bg-gray-700');
                });
                tabs[key].tab.classList.remove('hidden');
                tabs[key].btn.classList.add('active', 'bg-amber-600');
                tabs[key].btn.classList.remove('bg-gray-700');
                updateUI(); // Ensure crafting list is populated
            });
        });

        // Sub-Tab Switching
        const characterSubtabs = {
            inventory: { tab: document.getElementById('inventory-content'), btn: document.getElementById('subtab-inventory') },
            gear: { tab: document.getElementById('gear-content'), btn: document.getElementById('subtab-gear') },
            battle: { tab: document.getElementById('battle-content'), btn: document.getElementById('subtab-battle') }
        };
        Object.keys(characterSubtabs).forEach(key => {
            characterSubtabs[key].btn.addEventListener('click', () => {
                Object.values(characterSubtabs).forEach(t => {
                    t.tab.classList.add('hidden');
                    t.btn.classList.remove('active', 'bg-green-600');
                    t.btn.classList.add('bg-gray-700');
                });
                characterSubtabs[key].tab.classList.remove('hidden');
                characterSubtabs[key].btn.classList.add('active', 'bg-green-600');
                characterSubtabs[key].btn.classList.remove('bg-gray-700');
            });
        });

        // Event Listeners
        ['wood', 'stone', 'hemp', 'fat', 'bone', 'ore', 'berry', 'fish'].forEach(resource => {
            document.getElementById(`gather-${resource}`).addEventListener('click', () => gatherResource(resource));
        });
        ['weapon', 'utility', 'armor', 'tool1', 'heal'].forEach(slot => {
            document.getElementById(`equip-${slot}`).addEventListener('click', () => equipItem(slot));
        });
        document.getElementById('use-item').addEventListener('click', useItem);
        document.getElementById('modal-close').addEventListener('click', () => document.getElementById('modal').style.display = 'none');

        // Passive Updates
        setInterval(() => {
            gameState.stats.hunger = Math.max(0, gameState.stats.hunger - 1);
            gameState.stats.energy = Math.min(50, gameState.stats.energy + 0.5);
            updateUI();
        }, 5000);

        // Initial Setup
        updateUI();
    </script>
