function log(s){ const d=document.createElement('div'); d.textContent = s; logEl.prepend(d); }
function clamp(v,a,b){ return Math.max(a,Math.min(b,v)); }
function randInt(a,b){ return Math.floor(Math.random()*(b-a+1))+a; }

/* compute hero final stats from race,class,alloc and level scaling */
function buildHeroRuntime(hero, lvl){
  // base numeric stats derived from attributes:
  // health = 100 + Vitality*12 + Strength*6
  // damage = 6 + Strength*2 + Dexterity*1.2 + Intellect*0.5
  // defense = 2 + Endurance*1.5 + Vitality*0.3
  // attackSpeed = 1 + Dexterity*0.03 + Agility*0.02
  // critChance = Dexterity*0.15
  // mana = 20 + Intellect*10 + Wisdom*4
  // apply level scaling: 5% per level to hp & damage modestly
  const r = RACES[hero.race].mod;
  const c = CLASSES[hero.class].base;
  // combine base points: allocated + race mod + class base
  const combined = {};
  ATTRS.forEach(a=>{
    const baseVal = (allocPrototypeValue(hero,a) || 0); // placeholder - hero was created with a snapshot
    combined[a] = (hero.alloc && hero.alloc[a] ? hero.alloc[a] : 0) + (r[a]||0) + (c[a]||0);
  });
  const hp = Math.max(30, Math.floor(100 + combined.Vitality*12 + combined.Strength*6) * (1 + lvl*0.05));
  const damage = Number(((6 + combined.Strength*2 + combined.Dexterity*1.2 + combined.Intellect*0.5) * (1 + lvl*0.03)).toFixed(1));
  const defense = Number((2 + combined.Endurance*1.5 + combined.Vitality*0.3).toFixed(2));
  const attackSpeed = Number((1 + combined.Dexterity*0.03 + combined.Agility*0.02).toFixed(3));
  const crit = Number((combined.Dexterity*0.15).toFixed(2));
  const mana = Math.max(0, Math.floor(20 + combined.Intellect*10 + combined.Wisdom*4));
  return {
    maxHp: hp, hp: hp, dmg: damage, def: defense, spd: attackSpeed, crit: crit, mana: mana,
    name: hero.name, race:hero.race, class:hero.class, weapon:hero.weapon, color: hero.color || '#8ab4ff',
    skill: WEAPONS[hero.weapon].skill(hero), cooldown:0, alive:true
  };
}
function allocPrototypeValue(hero, attr){ return 0; } // not used; hero stores alloc at creation

}

/* Enemy generator based on level */
const ENEMIES = [
  { id:'bandit', name:'Bandit', base:{hp:120,dmg:14,def:2,spd:10}, label:'bandit' },
  { id:'savage', name:'Savage Beast', base:{hp:260,dmg:26,def:5,spd:12}, label:'beast' },
  { id:'pirate', name:'Pirate Captain', base:{hp:420,dmg:40,def:8,spd:11}, label:'captain' }
];
function spawnEnemyForLevel(lvl){
  const type = lvl<=6 ? ENEMIES[0] : (lvl<=13 ? ENEMIES[1] : ENEMIES[2]);
  const mult = 1 + lvl*0.08;
  const e = {
    id: type.id,
    name: type.name,
    maxHp: Math.floor((type.base.hp + lvl*30)*mult),
    hp: Math.floor((type.base.hp + lvl*30)*mult),
    dmg: Number(((type.base.dmg + lvl*4)*mult).toFixed(1)),
    def: Number(((type.base.def + lvl*0.6)*mult).toFixed(2)),
    spd: Math.max(4, Math.round(type.base.spd + lvl*0.2)),
    alive: true,
    color:'#ef4444'
  };
  return e;
}

/* Battle setup & loop */
let animQueue = [];
function buildCombatants(){
  const heroes = lineup.map(id=>{
    const h = roster.find(x=>x.id===id);
    const runt = buildHeroRuntime(h, level);
    // override skill based on chosen weapon (use hero-specific alloc snapshot)
    runt.skill = WEAPONS[h.weapon].skill(h);
    runt.name = h.name;
    return runt;
  });
  const enemy = spawnEnemyForLevel(level);
  // position y
  heroes.forEach((hh,i)=> hh.y = 80 + i*90, hh.x=100);
  enemy.x=680; enemy.y=160;
  combat.heroes = heroes; combat.enemy = enemy; updateHPBars(); renderCanvas(); renderEnemyInfo();
}

function startBattle(){
  if(lineup.length===0){ alert('Add at least one hero'); return; }
  if(combat.running) return;
  log('Battle started');
  buildCombatants();
  combat.running = true;
  loopCombat();
}

async function loopCombat(){
  while(combat.running){
    // build order by speed
    const actors = [...combat.heroes.filter(h=>h.alive), combat.enemy.alive ? [combat.enemy] : []].flat();
    const order = [...combat.heroes.filter(h=>h.alive), combat.enemy].sort((a,b)=>b.spd - a.spd);
    for(const actor of order){
      if(!combat.running) break;
      if(actor === combat.enemy){
        if(!actor.alive) continue;
        await delay(450);
        enemyAct();
        if(checkEnd()) break;
      } else {
        if(!actor.alive) continue;
        await delay(300);
        heroAct(actor);
        if(checkEnd()) break;
      }
    }
    await delay(200);
    // loop end
    if(!combat.running) break;
    // if enemy dead or all heroes dead, exit loop
    if(checkEnd()) break;
    // continue next round
  }
}

function heroAct(h){
  // use skill if cooldown 0 and enough mana
  if(h.skill && h.cooldown<=0 && h.mana >= h.skill.cost){
    h.mana -= h.skill.cost; h.cooldown = h.skill.cd;
    const dmg = calcDamage(h.dmg * h.skill.dmgMult, h, combat.enemy);
    combat.enemy.hp -= dmg;
    log(`${h.name} used ${h.skill.name} for ${dmg} dmg`);
    animateAttack(h, combat.enemy);
  } else {
    const dmg = calcDamage(h.dmg, h, combat.enemy);
    combat.enemy.hp -= dmg;
    log(`${h.name} attacked for ${dmg} dmg`);
    animateAttack(h, combat.enemy);
  }
  // reduce cooldowns
  combat.heroes.forEach(x=>{ if(x.cooldown>0) x.cooldown = Math.max(0, x.cooldown - 1); });
  updateHPBars(); renderEnemyInfo();
}

function enemyAct(){
  const e = combat.enemy;
  const targets = combat.heroes.filter(h=>h.alive);
  if(targets.length===0) return;
  const t = targets[randInt(0,targets.length-1)];
  const dmg = calcDamage(e.dmg, e, t);
  t.hp -= dmg;
  log(`${e.name} hit ${t.name} for ${dmg} dmg`);
  animateAttack(e,t);
  updateHPBars();
}

function calcDamage(raw, attacker, defender){
  const mitig = defender.def * 0.7;
  return Math.max(1, Math.round(raw - mitig));
}

function checkEnd(){
  // clamp
  if(combat.enemy.hp <= 0){
    combat.enemy.alive = false; combat.running=false;
    log(`${combat.enemy.name} defeated!`);
    updateHPBars(); renderEnemyInfo();
    return true;
  }
  const alive = combat.heroes.some(h=>h.hp>0);
  if(!alive){
    combat.heroes.forEach(h=>h.alive=false); combat.running=false;
    log('All heroes down — defeat.');
    updateHPBars(); return true;
  }
  return false;
}

/* Animation & canvas */
function animateAttack(a,b){
  animQueue.push({att:a,tgt:b,progress:0});
  requestAnimationFrame(animTick);
}
function animTick(){
  animQueue = animQueue.filter(anim=>{
    anim.progress += 0.09;
    return anim.progress < 1;
  });
  renderCanvas();
  if(animQueue.length>0) requestAnimationFrame(animTick);
}
function renderCanvas(){
  ctx.clearRect(0,0,canvas.width,canvas.height);
  // grid faint
  ctx.save(); ctx.globalAlpha=0.03; ctx.fillStyle='#fff';
  for(let x=0;x<canvas.width;x+=40) ctx.fillRect(x,0,1,canvas.height);
  for(let y=0;y<canvas.height;y+=40) ctx.fillRect(0,y,canvas.width,1); ctx.restore();
  // draw heroes
  if(combat.heroes) combat.heroes.forEach(h=>drawActor(h));
  if(combat.enemy) drawActor(combat.enemy,true);
  animQueue.forEach(a=>{
    const p=a.progress; const sx=a.att.x, sy=a.att.y, tx=a.tgt.x, ty=a.tgt.y;
    const mx = sx + (tx-sx)*Math.min(1,p*2); const my = sy + (ty-sy)*Math.min(1,p*2);
    ctx.beginPath(); ctx.fillStyle='#ffd166'; ctx.arc(mx,my,8,0,Math.PI*2); ctx.fill();
  });
}
function drawActor(actor,isEnemy){
  const x = actor.x||100, y = actor.y||160;
  ctx.beginPath(); ctx.fillStyle = actor.color||'#888'; ctx.roundRect ? ctx.roundRect(x-36,y-36,72,72,10) : ctx.fillRect(x-36,y-36,72,72); ctx.fill();
  ctx.fillStyle='#fff'; ctx.font='14px Inter'; ctx.textAlign='center'; ctx.fillText(actor.name, x, y-46);
  // hp bar
  const pct = actor.maxHp ? clamp(Math.floor(actor.hp*100/actor.maxHp),0,100) : 0;
  ctx.fillStyle='#1f2937'; ctx.fillRect(x-40,y+42,80,10);
  ctx.fillStyle = isEnemy? '#ef4444' : '#10b981'; ctx.fillRect(x-40,y+42,80*pct/100,10);
  ctx.fillStyle='#e6eef8'; ctx.font='12px Inter'; ctx.fillText(`${Math.max(0,Math.floor(actor.hp))}/${actor.maxHp}`, x, y+60);
  if(!actor.alive || actor.hp<=0){ ctx.fillStyle='rgba(0,0,0,0.6)'; ctx.fillRect(x-36,y-36,72,72); ctx.fillStyle='#ef4444'; ctx.fillText('DEAD', x, y+6); }
}
if(!CanvasRenderingContext2D.prototype.roundRect){
  CanvasRenderingContext2D.prototype.roundRect = function(x,y,w,h,r){ this.beginPath(); this.moveTo(x+r,y); this.arcTo(x+w,y,x+w,y+h,r); this.arcTo(x+w,y+h,x,y+h,r); this.arcTo(x,y+h,x,y,r); this.arcTo(x,y,x+w,y,r); this.closePath(); this.fill(); };
}

/* HP bars & enemy info update */
function updateHPBars(){
  hpBars.innerHTML = '';
  if(!combat.heroes || combat.heroes.length===0){ hpBars.innerHTML = '<div class="small">No heroes</div>'; return; }
  combat.heroes.forEach(h=>{
    const pct = h.maxHp ? Math.max(0,Math.floor(h.hp*100/h.maxHp)) : 0;
    const div = document.createElement('div'); div.style.marginBottom='8px';
    div.innerHTML = `<div style="display:flex;justify-content:space-between"><strong style="color:${h.color}">${h.name}</strong><span class="small">${Math.max(0,Math.floor(h.hp))}/${h.maxHp}</span></div>
      <div class="bar" style="margin-top:4px"><i style="width:${pct}%"></i></div>`;
    hpBars.appendChild(div);
  });
}
function renderEnemyInfo(){
  if(!combat.enemy){ enemyInfo.textContent = 'No enemy'; enemyBar.innerHTML=''; return; }
  const e = combat.enemy; enemyInfo.innerHTML = `<strong style="color:#ef4444">${e.name}</strong> • HP ${e.hp}/${e.maxHp} • DMG ${e.dmg}`;
  const pct = Math.max(0, Math.floor(e.hp*100/e.maxHp));
  enemyBar.innerHTML = `<div class="bar"><i style="width:${pct}%"></i></div>`;
}
